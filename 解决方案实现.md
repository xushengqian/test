# FreeSWITCH DESTINATION_OUT_OF_ORDER 解决方案实现

## 1. 错误处理和重试机制

### Java 代码示例

```java
public class FreeSwitchClientService {
    
    private static final int MAX_RETRY_COUNT = 3;
    private static final long RETRY_DELAY_MS = 1000;
    
    /**
     * 处理 originate 命令，带重试机制
     */
    public void originateWithRetry(OriginateRequest request) {
        int retryCount = 0;
        Exception lastException = null;
        
        while (retryCount < MAX_RETRY_COUNT) {
            try {
                String result = executeOriginate(request);
                
                if (isSuccess(result)) {
                    return;
                }
                
                // 检查是否是 DESTINATION_OUT_OF_ORDER 错误
                if (isDestinationOutOfOrderError(result)) {
                    log.warn("DESTINATION_OUT_OF_ORDER 错误，尝试重试. 重试次数: {}", retryCount);
                    
                    // 尝试切换到备用网关
                    if (retryCount > 0) {
                        request = switchToBackupGateway(request);
                    }
                    
                    // 等待后重试
                    Thread.sleep(RETRY_DELAY_MS * (retryCount + 1));
                    retryCount++;
                    continue;
                }
                
                // 其他错误，直接抛出
                throw new FreeSwitchException("Originate 失败: " + result);
                
            } catch (Exception e) {
                lastException = e;
                retryCount++;
                
                if (retryCount >= MAX_RETRY_COUNT) {
                    break;
                }
                
                log.warn("Originate 执行异常，准备重试. 重试次数: {}", retryCount, e);
                try {
                    Thread.sleep(RETRY_DELAY_MS * retryCount);
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    throw new RuntimeException(ie);
                }
            }
        }
        
        // 所有重试都失败
        throw new FreeSwitchException("Originate 重试失败，已重试 " + MAX_RETRY_COUNT + " 次", lastException);
    }
    
    private boolean isDestinationOutOfOrderError(String result) {
        return result != null && result.contains("DESTINATION_OUT_OF_ORDER");
    }
    
    private boolean isSuccess(String result) {
        return result != null && !result.startsWith("-ERR");
    }
    
    private OriginateRequest switchToBackupGateway(OriginateRequest request) {
        // 切换到备用网关逻辑
        String backupGateway = getBackupGateway(request.getGateway());
        request.setGateway(backupGateway);
        return request;
    }
}
```

## 2. 连接池管理

```java
public class FreeSwitchConnectionPool {
    
    private final Map<String, Queue<FreeSwitchConnection>> connectionPools;
    private final int maxConnectionsPerGateway;
    private final int maxTotalConnections;
    
    public FreeSwitchConnection getConnection(String gateway) {
        Queue<FreeSwitchConnection> pool = connectionPools.get(gateway);
        
        if (pool == null || pool.isEmpty()) {
            if (canCreateNewConnection(gateway)) {
                return createNewConnection(gateway);
            } else {
                // 连接池已满，等待或抛出异常
                throw new ConnectionPoolExhaustedException("网关连接池已满: " + gateway);
            }
        }
        
        FreeSwitchConnection conn = pool.poll();
        
        // 检查连接是否健康
        if (!conn.isHealthy()) {
            conn.close();
            return getConnection(gateway); // 递归获取新连接
        }
        
        return conn;
    }
    
    public void returnConnection(FreeSwitchConnection conn) {
        if (conn.isHealthy()) {
            Queue<FreeSwitchConnection> pool = connectionPools.get(conn.getGateway());
            if (pool != null && pool.size() < maxConnectionsPerGateway) {
                pool.offer(conn);
            } else {
                conn.close();
            }
        } else {
            conn.close();
        }
    }
}
```

## 3. 负载均衡和健康检查

```java
public class FreeSwitchLoadBalancer {
    
    private final List<FreeSwitchServer> servers;
    private final HealthChecker healthChecker;
    
    public FreeSwitchServer selectServer() {
        // 过滤健康的服务器
        List<FreeSwitchServer> healthyServers = servers.stream()
            .filter(server -> healthChecker.isHealthy(server))
            .collect(Collectors.toList());
        
        if (healthyServers.isEmpty()) {
            throw new NoHealthyServerException("没有可用的健康服务器");
        }
        
        // 基于负载选择服务器（轮询、最少连接等）
        return selectByLeastConnections(healthyServers);
    }
    
    private FreeSwitchServer selectByLeastConnections(List<FreeSwitchServer> servers) {
        return servers.stream()
            .min(Comparator.comparing(FreeSwitchServer::getActiveConnections))
            .orElse(servers.get(0));
    }
}
```

## 4. 监控和告警

```java
@Component
public class FreeSwitchMonitor {
    
    private final MeterRegistry meterRegistry;
    
    @EventListener
    public void onBackgroundJobResult(BackgroundJobResultEvent event) {
        String result = event.getResult();
        
        // 记录指标
        if (isDestinationOutOfOrderError(result)) {
            meterRegistry.counter("freeswitch.error.destination_out_of_order",
                "server", event.getServer(),
                "gateway", event.getGateway()).increment();
            
            // 检查是否需要告警
            checkAndAlert(event);
        }
    }
    
    private void checkAndAlert(BackgroundJobResultEvent event) {
        long errorCount = getErrorCount(event.getServer(), event.getGateway());
        
        if (errorCount > getErrorThreshold()) {
            alertService.sendAlert("FreeSWITCH 错误率过高", 
                String.format("服务器 %s 网关 %s 错误数: %d", 
                    event.getServer(), event.getGateway(), errorCount));
        }
    }
}
```

## 5. 网关状态检查

```java
public class GatewayHealthChecker {
    
    /**
     * 检查网关状态
     */
    public boolean checkGatewayStatus(String gateway) {
        try {
            String command = "sofia status gateway " + gateway;
            String result = executeCommand(command);
            
            // 解析结果，检查网关是否注册成功
            return parseGatewayStatus(result);
        } catch (Exception e) {
            log.error("检查网关状态失败: " + gateway, e);
            return false;
        }
    }
    
    /**
     * 重新注册网关
     */
    public void reregisterGateway(String gateway) {
        try {
            String command = "sofia profile external restart " + gateway;
            executeCommand(command);
            log.info("网关重新注册: " + gateway);
        } catch (Exception e) {
            log.error("网关重新注册失败: " + gateway, e);
        }
    }
}
```

```java
@Aspect
@Component
public class FreeSwitchLoggingAspect {
    
    @Around("execution(* FreeSwitchClientService.originate*(..))")
    public Object logOriginate(ProceedingJoinPoint joinPoint) throws Throwable {
        long startTime = System.currentTimeMillis();
        Object[] args = joinPoint.getArgs();
        
        try {
            Object result = joinPoint.proceed();
            long duration = System.currentTimeMillis() - startTime;
            
            log.info("Originate 执行成功, 耗时: {}ms, 参数: {}", duration, args);
            return result;
            
        } catch (Exception e) {
            long duration = System.currentTimeMillis() - startTime;
            log.error("Originate 执行失败, 耗时: {}ms, 参数: {}", duration, args, e);
            throw e;
        }
    }
}
```
